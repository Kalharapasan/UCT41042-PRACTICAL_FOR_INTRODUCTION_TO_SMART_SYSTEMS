create table public.street_light_logs (
  id bigint generated by default as identity not null,
  intensity integer not null,
  status character varying null,
  "ledState" character varying null,
  created_at timestamp without time zone null default now(),
  constraint street_light_logs_pkey primary key (id)
) TABLESPACE pg_default;

create trigger update_test_updated_at BEFORE
update on street_light_logs for EACH row
execute FUNCTION update_updated_at_column ();
